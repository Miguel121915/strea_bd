<script>
// Videos de Hardstyle que permiten reproducción externa
const videos = [
    { id: 'kDlTSynqMTs', titulo: 'Defqon.1 2024 | Official Aftermovie' },
    { id: '5NHgoRQDUH0', titulo: 'Defqon.1 2025 | The Endshow' },
    { id: 'fN1Gw8PHkAI', titulo: 'Defqon.1 2013 | Endshow Saturday' },
    { id: 'DpxyDPzQZ7E', titulo: 'Defqon.1 2018 | Official Saturday Endshow' },
    { id: 'ajTMOR7Ke_I', titulo: 'Defqon.1 2023 | Aftermovie: Path of the Warrior' }
];

let videoActivo = 0;

function renderPlaylist(){
    const lista = document.getElementById('playlistLista');
    lista.innerHTML = '';
    videos.forEach((v, i) => {
        const item = document.createElement('div');
        item.className = 'playlist-item' + (i === videoActivo ? ' activo' : '');
        item.innerHTML = `
            <img class="playlist-thumb" src="https://img.youtube.com/vi/${v.id}/mqdefault.jpg">
            <div class="item-info">
                <div class="item-titulo">${v.titulo}</div>
            </div>`;
        item.onclick = () => cambiarVideo(i);
        lista.appendChild(item);
    });
}

function cambiarVideo(index){
    videoActivo = index;
    const v = videos[index];
    document.getElementById('videoFrame').src = 'https://www.youtube.com/embed/' + v.id + '?autoplay=1';
    document.getElementById('tituloVideo').innerText = v.titulo;
    renderPlaylist();
}

function registrarUsuario(){
    const nombre = document.getElementById('nombre').value;
    const fecha = document.getElementById('fechaNac').value;
    const numero = document.getElementById('numero').value;

    if(!nombre || !fecha || !numero){ alert('Por favor, llena todos los campos'); return; }

    const barraBox = document.getElementById('barraBox');
    const barra = document.getElementById('barra');
    barraBox.style.display = 'block';
    
    let progreso = 0;
    const intervalo = setInterval(() => {
        progreso += 5;
        barra.value = progreso;
        if(progreso >= 100){
            clearInterval(intervalo);
            barraBox.style.display = 'none';
            const u = { nombre, fecha, numero };
            guardarUsuario(u);
            mostrarUsuario(u);
            limpiarFormulario();
        }
    }, 30);
}

function guardarUsuario(u){
    const arr = JSON.parse(localStorage.getItem('usuarios_pension')) || [];
    arr.push(u);
    localStorage.setItem('usuarios_pension', JSON.stringify(arr));
}

function mostrarUsuario(u){
    const res = document.getElementById('resultado');
    const t = document.createElement('div');
    t.className = 'tarjeta';
    t.innerHTML = `<b>NUEVO PERFIL</b><br>Nombre: ${u.nombre}<br>F. Nacimiento: ${u.fecha}<br>Tel: ${u.numero}`;
    res.prepend(t);
}

function limpiarFormulario(){
    ['nombre','fechaNac','numero'].forEach(id => document.getElementById(id).value = '');
}

window.onload = () => {
    cambiarVideo(0);
    const u = JSON.parse(localStorage.getItem('usuarios_pension')) || [];
    u.forEach(user => mostrarUsuario(user));
};
